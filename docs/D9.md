```sh
# see usr/bin/check_silent_upgrade.sh
# see ava/script/ota_unpack.sh
# ava/script/ota_dw_fw.sh

ota_decrypt(){
  FW_FILE=${FW_FILE:-d82bfa4f4bcce2f5da4410403fbc1d65_upd_dreame.vacuum.p2009.bin}
  OTA_PASSWORD=${OTA_PASSWORD:-NDVlNDJlMjk3NTY0MjU4OGU0YzFiMDBjNjMyNzM2M2QgIC0K}

  [ -e "${FW_FILE}" ] || return 0
  [ -e publickey.pem ] || return 0
  [ -e OTA_Key_pub.pem ] || return 0

  openssl smime -decrypt \
    -in ${FW_FILE} \
    -binary -inform DEM \
    -inkey publickey.pem \
    -out update.zip

  unzip -P "${OTA_PASSWORD}" update.zip -d tmp
  openssl dgst -sha256 -verify OTA_Key_pub.pem -signature tmp/firmware.signature tmp/update.img
}

ota_extract(){
  
  if [ ! -x afptool ]; then
    curl -LO https://github.com/suyulin/apftool-rs/releases/download/v1.1.0/afptool-rs-linux-x86_64.zip
    unzip afptool-rs-linux-x86_64.zip
    mv afptool-rs-linux-x86_64 afptool
  fi

  [ -x ./afptool ] || return 0
  ./afptool unpack tmp/update.img extract
}

ota_decrypt
ota_extract
```
